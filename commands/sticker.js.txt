import fetch from 'node-fetch';

export default {
  name: 'stickers',
  description: 'Busca stickers',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /stickers [texto]');
    const query = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/s/stickers?q=${encodeURIComponent(query)}&apikey=lem336`);
      const data = await res.json();
      if (data?.stickers) await bot.sendSticker(chatId, data.stickers[0].url);
      else await bot.sendMessage(chatId, '❌ No encontré stickers.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al buscar stickers.');
    }
  }
};
