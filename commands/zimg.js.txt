import fetch from 'node-fetch';

export default {
  name: 'zimg',
  description: 'Genera imágenes con IA (Zimg)',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /zimg [descripción]');
    const prompt = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/ai/zimg?size=1024x1024&q=${encodeURIComponent(prompt)}&apikey=lem336`);
      const buffer = await res.buffer();
      await bot.sendPhoto(chatId, buffer, { caption: `🖼️ "${prompt}"` });
    } catch {
      await bot.sendMessage(chatId, '❌ Error al generar imagen.');
    }
  }
};
