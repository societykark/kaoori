import fetch from 'node-fetch';

export default {
  name: 'qwen',
  description: 'Pregunta a Qwen (IA)',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /qwen [pregunta]');
    const prompt = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/ai/qwen?q=${encodeURIComponent(prompt)}&apikey=lem336`);
      const data = await res.json();
      if (data?.result) await bot.sendMessage(chatId, `📡 *Qwen:*\n${data.result}`);
      else await bot.sendMessage(chatId, '❌ No pude obtener respuesta de Qwen.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al consultar Qwen.');
    }
  }
};
