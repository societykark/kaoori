import fetch from 'node-fetch';

export default {
  name: 'facebook',
  description: 'Descarga videos de Facebook',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /facebook [url]');
    const url = args[0];
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/dl/facebook?url=${encodeURIComponent(url)}&quality=hd&apikey=lem336`);
      const data = await res.json();
      if (data?.download_url) await bot.sendVideo(chatId, data.download_url);
      else await bot.sendMessage(chatId, '❌ No se pudo descargar el video.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al descargar.');
    }
  }
};
