import fetch from 'node-fetch';

export default {
  name: 'claude',
  description: 'Pregunta a Claude (IA)',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /claude [pregunta]');
    const prompt = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/ai/claude?q=${encodeURIComponent(prompt)}&apikey=lem336`);
      const data = await res.json();
      if (data?.result) await bot.sendMessage(chatId, `🤖 *Claude:*\n${data.result}`);
      else await bot.sendMessage(chatId, '❌ No pude obtener respuesta de Claude.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al consultar Claude.');
    }
  }
};
