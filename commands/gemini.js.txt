import fetch from 'node-fetch';

export default {
  name: 'gemini',
  description: 'Pregunta a Gemini (IA)',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /gemini [pregunta]');
    const prompt = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/ai/gemini?q=${encodeURIComponent(prompt)}&apikey=lem336`);
      const data = await res.json();
      if (data?.result) await bot.sendMessage(chatId, `🧠 *Gemini:*\n${data.result}`);
      else await bot.sendMessage(chatId, '❌ No pude obtener respuesta de Gemini.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al consultar Gemini.');
    }
  }
};
