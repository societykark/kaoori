import fetch from 'node-fetch';

export default {
  name: 'yta',
  description: 'Descarga audio de YouTube',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /yta [url o búsqueda]');
    const query = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/dl/yta?q=${encodeURIComponent(query)}&apikey=lem336`);
      const data = await res.json();
      if (data?.download_url) await bot.sendAudio(chatId, data.download_url);
      else await bot.sendMessage(chatId, '❌ No se pudo descargar el audio.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al descargar.');
    }
  }
};
