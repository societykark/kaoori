import fetch from 'node-fetch';

export default {
  name: 'tiktoksearch',
  description: 'Busca videos en TikTok',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /tiktoksearch [texto]');
    const query = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/s/tiktok?q=${encodeURIComponent(query)}&apikey=lem336`);
      const data = await res.json();
      if (data?.videos) await bot.sendVideo(chatId, data.videos[0].url, { caption: `🎵 ${data.videos[0].title}` });
      else await bot.sendMessage(chatId, '❌ No encontré videos.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al buscar en TikTok.');
    }
  }
};
