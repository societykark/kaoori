import fetch from 'node-fetch';

export default {
  name: 'sp',
  description: 'Busca canciones en Spotify',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /sp [canción]');
    const query = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/s/sp?q=${encodeURIComponent(query)}&limit=10&apikey=lem336`);
      const data = await res.json();
      if (data?.tracks) {
        let msg = '🎵 *Resultados en Spotify:*\n\n';
        data.tracks.slice(0, 5).forEach((t, i) => {
          msg += `${i+1}. *${t.name}* - ${t.artists[0].name}\n🔗 ${t.external_urls.spotify}\n\n`;
        });
        await bot.sendMessage(chatId, msg);
      } else await bot.sendMessage(chatId, '❌ No encontré canciones.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al buscar en Spotify.');
    }
  }
};
