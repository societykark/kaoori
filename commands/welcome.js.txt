import fetch from 'node-fetch';

export default {
  name: 'welcome',
  description: 'Genera imagen de bienvenida',
  async execute({ bot, chatId, args }) {
    const username = args[0] || 'Usuario';
    const guildName = args[1] || 'Club de Anime';
    await bot.sendChatAction(chatId, 'typing');
    try {
      const url = `https://api.lempi.lat/api/canvas/welcomev1?username=${encodeURIComponent(username)}&guildName=${encodeURIComponent(guildName)}&guildIcon=https%3A%2F%2Fapi.lempi.lat%2FUka.jpg&memberCount=150&avatar=https%3A%2F%2Fapi.lempi.lat%2FBiy.jpg&background=https%3A%2F%2Fapi.lempi.lat%2FRlK.jpg&quality=80&apikey=lem336`;
      const buffer = await fetch(url).then(r => r.buffer());
      await bot.sendPhoto(chatId, buffer, { caption: `🎉 Bienvenido ${username} a ${guildName}!` });
    } catch {
      await bot.sendMessage(chatId, '❌ Error al generar la imagen de bienvenida.');
    }
  }
};
