import fetch from 'node-fetch';

export default {
  name: 'whatmusic',
  description: 'Identifica una canción',
  async execute({ bot, chatId, args }) {
    if (!args.length) return bot.sendMessage(chatId, '❌ Uso: /whatmusic [url o nombre]');
    const query = args.join(' ');
    await bot.sendChatAction(chatId, 'typing');
    try {
      const res = await fetch(`https://api.lempi.lat/tools/whatmusic?q=${encodeURIComponent(query)}&apikey=lem336`);
      const data = await res.json();
      if (data?.result) await bot.sendMessage(chatId, `🎵 *Canción identificada:*\n${data.result}`);
      else await bot.sendMessage(chatId, '❌ No se pudo identificar la canción.');
    } catch {
      await bot.sendMessage(chatId, '❌ Error al identificar.');
    }
  }
};
